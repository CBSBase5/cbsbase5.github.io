   web app (Execute as: Me; Who has access: anyone in the school).

   hidden page (the bridge, ?bridge=1) that opens inside the site

               each student's code; signed in, pages open by themselves

               a Date still to come keeps it back until that day)
     Sends     everything students send to Jay, newest at the bottom

     BridgeLink  Done ticks from the BridgeLink copy list
               (/teacher/bridgelink.html); made the first time one is ticked
     BridgeLink names  BridgeLink's name for each target, learned by the
               copy list from a pasted BridgeLink page

   Nothing here asks for names: students are their initials.
   ========================================================== */

function doGet(e){
  var p = (e && e.parameter) || {};
  if(p.bridge) return HtmlService.createHtmlOutput(bridgeHtml_())
    .setTitle("Base 5")
    .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL);

function doPost(e){
  var b = {};
  try{ b = JSON.parse((e && e.postData && e.postData.contents) || "{}"); }catch(err){ return out_({ ok: false, error: "bad body" }); }
  return out_(api(b));
}

/* every request, from the bridge or called directly */
function api(r){
  r = r || {};
  try{
    var a = r.action;
    if(a === "ping") return { ok: true };
    if(a === "load") return load_(who_(r.who), +r.since || 0);
    if(a === "folder") return folder_(who_(r.who));
    if(a === "overview") return overview_(r.key);
    if(a === "whoami") return whoami_();
    if(a === "codes") return codes_(r.key);
    var read3 = READ3[a];
    if(read3) return read3(r);
    if(WRITE3[a]){
      var lock3 = LockService.getScriptLock();
      lock3.waitLock(15000);
      try{ return WRITE3[a](r); }finally{ try{ lock3.releaseLock(); }catch(e3){} }
    }
    if(a === "save" || a === "send" || a === "seen"){
      var lock = LockService.getScriptLock();
      lock.waitLock(15000);
      try{
        if(a === "save") return save_(who_(r.who), r.items || {});
        if(a === "send") return send_(r);
        return seen_(r.key, +r.row, r.seen !== false);
      }finally{ try{ lock.releaseLock(); }catch(e2){} }
    }
    return { ok: false, error: "unknown action" };
  }catch(err){ return { ok: false, error: String(err.message || err) }; }
}
/* the bridge page calls this with google.script.run; text in, text out */
function apiJson(req){
  var r = {};
  try{ r = typeof req === "string" ? JSON.parse(req) : (req || {}); }catch(e){}
  return JSON.stringify(api(r));
}

function bridgeHtml_(){
  return '<!DOCTYPE html><html><head><meta charset="utf-8"></head><body><script>' +
    'var SITE = ' + JSON.stringify(SITE) + ';' +
    'window.addEventListener("message", function(e){' +
    '  if(e.origin !== SITE) return;' +
    '  var m = e.data; if(!m || m.b5 !== "req") return;' +
    '  var reply = function(body){ e.source.postMessage({ b5: "res", id: m.id, body: body }, e.origin); };' +
    '  google.script.run' +
    '    .withSuccessHandler(reply)' +
    '    .withFailureHandler(function(err){ reply(JSON.stringify({ ok: false, error: String((err && err.message) || err) })); })' +
    '    .apiJson(JSON.stringify(m.req || {}));' +
    '});' +
    'try{ window.top.postMessage({ b5: "ready" }, SITE); }catch(x){}' +
    '<\/script></body></html>';
}

/* text that starts like a formula is kept as plain text */
function plain_(s){
  s = String(s == null ? "" : s);
  return /^[=+\-@]/.test(s) ? "'" + s : s;

/* ---------- progress ---------- */
function load_(who, since){
  var v = sheet_("Progress").getDataRange().getValues(), items = {};
  for(var i = 1; i < v.length; i++){
    if(v[i][0] !== who) continue;
    var t = +v[i][3] || 0;
    if(t > since) items[v[i][1]] = { v: String(v[i][2]), t: t };
  }
  return { ok: true, items: items, now: Date.now() };
}

function save_(who, items){
  var sh = sheet_("Progress"), v = sh.getDataRange().getValues(), at = {};
  for(var i = 1; i < v.length; i++) if(v[i][0] === who) at[v[i][1]] = i + 1;
  try{ fuelBaseline_(who, v); }catch(e0){}
  var events = [];
  var add = [], changed = 0;
  Object.keys(items).forEach(function(k){
    var it = items[k] || {}, val = it.v == null ? "" : String(it.v), t = +it.t || Date.now();
    if(k.length > 200 || val.length > MAX_CELL) return;
    var r = at[k];
    if(r){
      if(t < (+v[r - 1][3] || 0)) return;            /* something newer is already here */
      try{ events = events.concat(events_(who, k, String(v[r - 1][2]), val)); }catch(e1){}
      try{ evidenceFor_(who, k, val); }catch(ev1){}
      sh.getRange(r, 3, 1, 2).setValues([[val, t]]);
    } else {
      try{ events = events.concat(events_(who, k, "", val)); }catch(e2){}
      try{ evidenceFor_(who, k, val); }catch(ev2){}
      add.push([who, k, val, t]);
    }
    changed++;
  });
  if(add.length) sh.getRange(sh.getLastRow() + 1, 1, add.length, 4).setValues(add);
  if(events.length) logEvents_(who, events);
  return { ok: true, saved: changed, now: Date.now() };
}

/* ---------- sends to Jay ---------- */
function send_(b){
  var who = String(b.who || "").toUpperCase().replace(/[^A-Z0-9 ]/g, "").slice(0, 8) || "?";
  var what = String(b.tool || "Something").slice(0, 120);
  var page = String(b.page || "").slice(0, 200);
  var text = String(b.text || "").slice(0, MAX_CELL);
  var sh = sheet_("Sends");
  sh.appendRow([new Date(), plain_(who), plain_(page), plain_(what), plain_(text), ""]);
  if(WHO.indexOf(who) >= 0) logEvents_(who, [["Sent to Jay", what]]);
  if(what === "Idea for the site"){
    var idea = text.replace(/^An idea for the Base 5 site:\s*/, "").trim();
    if(idea && idea !== "(nothing typed yet)") sheet_("Ideas").appendRow([new Date(), plain_(idea.slice(0, 500)), "Thinking about it", "", 0, "", false, false]);
  }
  try{
    var on = String(setting_("Email me sends")).toUpperCase() === "TRUE", to = String(setting_("Email address")).trim();
    if(on && to) MailApp.sendEmail(to, "Base 5: " + who + " sent " + what, text + "\n\n(From " + page + ". Everything is in the Sends tab of your Base 5 sheet.)");
  }catch(err){}
  return { ok: true };
}

/* ---------- folders ---------- */
function folder_(who){
  var v = sheet_("Folders").getDataRange().getValues(), tz = Session.getScriptTimeZone();
  var out = { messages: [], activities: [], files: [], a2: false };
  var today = Utilities.formatDate(new Date(), tz, "yyyy-MM-dd");
  for(var i = 1; i < v.length; i++){
    var r = v[i], w = String(r[0]).trim();
    if(w !== who && w !== "Everyone") continue;
    var type = String(r[1]).trim(), title = String(r[3]).trim(), text = String(r[4]).trim(), note = String(r[5]).trim();
    if(/\/a2-[a-z0-9-]+\.html/.test(text)) out.a2 = true;      /* the Autumn 2 plan is in the sheet */
    if(r[6] === true || String(r[6]).toUpperCase() === "TRUE") continue;   /* ticked Hide */
    if(!type || (!title && !text)) continue;
    var date = r[2] instanceof Date ? Utilities.formatDate(r[2], tz, "yyyy-MM-dd") : String(r[2] || "").trim();
    if(date && date > today) continue;                            /* a date still to come: it waits until then */
    var id = "r" + digest_(w + "|" + type + "|" + title + "|" + text);
    if(type === "Message") out.messages.push({ id: id, date: date, text: text || title });
    else if(type === "File") out.files.push({ id: id, title: title || "A file", href: text, note: note, date: date });
    else out.activities.push({ id: id, title: title || "Something to try", href: text, note: note, date: date });
  }
  return { ok: true, who: who, data: out, now: Date.now() };
}

/* ---------- Autumn 2: put the term's drops into the Folders tab ----------
   Staff only. rows: [{who, date, title, href, note}]. Adds each row that
   is not there already (same student and same page), dated, so it shows
   in the folder from that day. Safe to run again. */
function a2plan_(r){
  staff_(r.key);
  var rows = Array.isArray(r.rows) ? r.rows : [];
  var sh = sheet_("Folders"), v = sh.getDataRange().getValues(), have = {};
  function base(t){ return String(t || "").trim().split("?")[0]; }
  for(var i = 1; i < v.length; i++) have[String(v[i][0]).trim() + "|" + base(v[i][4])] = 1;
  var add = [];
  rows.forEach(function(x){
    var who = String(x.who || "").toUpperCase(), href = String(x.href || "").trim();
    if(WHO.indexOf(who) < 0 || !/^\/[a-z0-9\-\/]+\/a2-[a-z0-9-]+\.html(\?lv=[a-z]+)?$/.test(href)) return;
    if(!/^\d{4}-\d{2}-\d{2}$/.test(String(x.date || ""))) return;
    if(have[who + "|" + base(href)]) return;
    have[who + "|" + base(href)] = 1;
    var p = String(x.date).split("-");
    add.push([who, "Something to try", new Date(+p[0], +p[1] - 1, +p[2]), clean_(x.title, 120), href, clean_(x.note || "", 600), false]);
  });
  if(add.length) sh.getRange(sh.getLastRow() + 1, 1, add.length, 7).setValues(add);
  return { ok: true, added: add.length, skipped: rows.length - add.length };
}

function digest_(s){
  var d = Utilities.computeDigest(Utilities.DigestAlgorithm.MD5, s, Utilities.Charset.UTF_8), h = "";
  for(var i = 0; i < 6; i++){ var b = (d[i] + 256) % 256; h += (b < 16 ? "0" : "") + b.toString(16); }
  return h;
}

/* ---------- who is signed in ---------- */
function signedIn_(){
  var email = "";
  try{ email = String(Session.getActiveUser().getEmail() || "").trim().toLowerCase(); }catch(e){}
  if(!email) return {};
  var sh = sheet_("People"); if(!sh) return {};
  var v = sh.getDataRange().getValues();
  for(var i = 1; i < v.length; i++){
    if(String(v[i][1]).trim().toLowerCase() !== email) continue;
    var w = String(v[i][0]).trim().toUpperCase();
    if(w === "STAFF") return { staff: true };
    if(WHO.indexOf(w) >= 0) return { who: w, code: String(v[i][2]).trim() };
  }
  return {};
}

function events_(who, key, oldV, newV){
  var P = PREFIX[who], out = [];
  if(!P || key.indexOf(P) !== 0) return out;
  var id = key.slice(P.length), o = parse_(oldV), n = parse_(newV);
  if(!n) return out;
  if(id === "hq.v1" && n.mood && n.mood.w){
    if(!o || !o.mood || o.mood.w !== n.mood.w || o.mood.d !== n.mood.d) out.push(["Check-in", MOODS[n.mood.w] || n.mood.w]);
  }
  if(/^[a-z0-9\-]+\.v1$/.test(id) && n.summary && n.summary.chunks){
    var act = n.summary.title ? String(n.summary.title) : nice_(id.replace(/\.v1$/, "")), os = (o && o.summary) || null, nd = n.summary.chunksDone || 0;
    if(!os) out.push(["Started", act]);
    if(nd > ((os && os.chunksDone) || 0)) out.push(["Chunk done", act + ": " + nd + " of " + n.summary.chunks]);
    if(n.summary.finished && !(os && os.finished)) out.push(["Finished", act]);
  }
  if(HQGAMES[key]){
    var nb = typeof n.best === "number" ? n.best : maxOf_(n.best), ob = o ? (typeof o.best === "number" ? o.best : maxOf_(o.best)) : 0;
    if(nb > ob && nb > 0) out.push(["New best", HQGAMES[key] + ": " + nb]);
  }
  return out;
}

/* The shared spark jar: sparks everyone in Base 5 has earned since Monday.
   The first baseline row for a student in a week is the real one. Each week's
   total is kept in Fuel weeks, so the jar can show the last few weeks too. */
function fuel_(){
  var wk = monday_(), pv = rows_("Progress"), fv = rows_("Fuel"), base = {}, total = 0, helpers = 0;
  for(var i = 1; i < fv.length; i++) if(dkey_(fv[i][0]) === wk && !(String(fv[i][1]) in base)) base[String(fv[i][1])] = +fv[i][2] || 0;
  WHO.forEach(function(w){
    if(!(w in base)) return;
    var got = Math.max(0, sparksOf_(w, pv) - base[w]);
    total += got; if(got > 0) helpers++;
  });
  var target = +setting_("Fuel target") || 300;
  var past = [];
  try{ past = fuelWeeks_(wk, total, target, helpers); }catch(e){}
  return { ok: true, week: wk, sparks: total, target: target, helpers: helpers, of: WHO.length,
    reward: String(setting_("Fuel reward") || ""), past: past };
}
/* writes this week's total (only when it changed) and hands back the four weeks before */
function fuelWeeks_(wk, total, target, helpers){
  var ss = SpreadsheetApp.getActive(), sh = ss.getSheetByName("Fuel weeks");
  if(!sh) return [];
  var v = sh.getDataRange().getValues(), row = 0, out = [];
  for(var i = 1; i < v.length; i++){
    var k = dkey_(v[i][0]);
    if(k === wk) row = i + 1;
    else if(k && k < wk) out.push({ week: k, sparks: +v[i][1] || 0, target: +v[i][2] || target, helpers: +v[i][3] || 0 });
  }
  var now = [wk, total, target, helpers, new Date()];
  if(!row) sh.appendRow(now);
  else if(+v[row - 1][1] !== total || +v[row - 1][2] !== target || +v[row - 1][3] !== helpers) sh.getRange(row, 1, 1, 5).setValues([now]);
  out.sort(function(a, b){ return a.week < b.week ? 1 : -1; });
  return out.slice(0, 4);
}

  }
  var all = Object.keys(best).filter(function(k){ return !hidden[k]; }).map(function(k){ return best[k]; })
    .sort(function(a, b){ return b.score - a.score || (a.base < b.base ? -1 : 1); });
  return { ok: true, top: all.slice(0, 3), also: all.slice(3).map(function(x){ return x.base; }).sort(),
    mine: hidden[me] ? null : (best[me] || null), count: all.length };

/* the same fingerprint as /data/b5core.js */
function fpNorm_(t){
  return String(t || "").toLowerCase().replace(/\u00d7/g, "x").replace(/labeled/g, "labelled").replace(/[\u2018\u2019']/g, "")
    .replace(/[^a-z0-9 ]+/g, " ").replace(/\s+/g, " ").trim();
}
function fpKey_(t){
  var s = fpNorm_(t), h = 2166136261, b = Utilities.newBlob(s).getBytes();
  for(var i = 0; i < b.length; i++){ h ^= (b[i] + 256) % 256; h = Math.imul(h, 16777619) >>> 0; }
  return h.toString(16);
}

/* what the activity was built for: { target, step, ms } */
function a2target_(who, id){
  var map = a2map_(), out = { target: "", step: 0, ms: null };
  var cov = map.drops[who + "|" + id];
  if(cov && cov.length){
    out.target = map.names[cov[0][0]] || cov[0][0]; out.step = +cov[0][1] || 0;
    try{ out.ms = buddyStep_(who, cov[0][0], out.step); }catch(e1){}
    return out;
  }
  var core = map.core[id];
  if(core){
    var rows = coreRows_().filter(function(r){ return r.who === who; }), best = null;
    rows.forEach(function(r){ if(!best && core.fp.indexOf(fpKey_(r.target)) >= 0) best = r; });

    });
    if(best) out.target = (best.level ? best.level + ": " : "Core target: ") + best.target;
  }
  return out;
}

  while(line.length < head.length) line.push("");
  var now = new Date(), stamp = Utilities.formatDate(now, tz_(), "yyyy-MM-dd HH:mm"), by = String(setting_("Evidence by") || "Base 5 site").trim();
  function put(k, val){ if(ci[k] >= 0) line[ci[k]] = val; }
  /* who worked on it with them: students to Students involved, Jay and HB to Staff involved */
  var withS = [], withT = [];
  String(e.with || "").split(/,\s*/).forEach(function(w){
    w = w.trim(); if(!w) return;
    if(WHO.indexOf(w.toUpperCase()) >= 0) withS.push(w.toUpperCase());
    else if(/^jay$/i.test(w)) withT.push("JR");
    else if(/^hb$/i.test(w)) withT.push("HB");
  });
  put("id", id); put("date", e.day); put("student", e.who); put("entry", e.note);
  put("staffIn", withT.join(", ")); put("others", withS.join(", "));
  if(!row){
    put("type", "Learning"); put("time", Utilities.formatDate(now, tz_(), "HH:mm")); put("tone", "Neutral"); put("lesson", e.subject);
    put("conf", ""); put("incident", "None"); put("safeguard", ""); put("ehcp", "");
    put("by", by); put("at", stamp); put("revision", 1); put("versions", 1);
  } else {
    put("editBy", by); put("editAt", stamp);
    if(ci.revision >= 0) put("revision", (+line[ci.revision] || 0) + 1);
  }
  if(e.msId && ci.milestones >= 0 && String(line[ci.milestones] || "").split(/\s*,\s*/).indexOf(e.msId) < 0) put("milestones", String(line[ci.milestones] || "") ? String(line[ci.milestones]) + "," + e.msId : e.msId);
  /* the picture they made: on the note like a photo, so Buddy offers it when the step's slide is made */
  if(e.picture && ci.photos >= 0){
    var ph = String(line[ci.photos] || "").split(",").map(function(x){ return x.trim(); }).filter(String);
    var pid = (e.picture.match(/[-\w]{25,}/) || [""])[0];
    if(pid && !ph.some(function(u){ return u.indexOf(pid) >= 0; })) ph.push(e.picture);
    put("photos", ph.join(","));
    if(ci.file >= 0 && !String(line[ci.file] || "")) put("file", e.picture);
  }
  /* games and practice are not built for one step, so Wrap up does not ask you to link them */
  if(!row && e.kind !== "activity" && !e.msId) put("linkSkip", "yes");
  /* Changed: left empty, so Buddy (1.22.1 and later) gives the note the next number in its own running count and
     its quick refresh picks it up. Older Buddys show it on their next full refresh. */
  put("changed", "");
  if(row) sh.getRange(row, 1, 1, line.length).setValues([line]);
  else sh.appendRow(line);
  return (row ? "updated " : "sent ") + Utilities.formatDate(now, tz_(), "HH:mm") + (e.msId ? " (linked to step)" : "");
}

/* ---------- who has been quiet on the site ----------
   School days (Monday to Friday) since each student last saved anything,
   from the Progress tab. Only for staff pages; students never see it. */
function schoolDaysSince_(t){
  if(!t) return 99;
  var d = new Date(t), now = new Date(), n = 0;
  d.setHours(12, 0, 0, 0); now.setHours(12, 0, 0, 0);
  while(d < now && n < 99){ d.setDate(d.getDate() + 1); var wd = d.getDay(); if(wd !== 0 && wd !== 6) n++; }
  return n;
}
function quiet_(min){
  var v = rows_("Progress"), last = {};
  for(var i = 1; i < v.length; i++){ var w = String(v[i][0]), t = +v[i][3] || 0; if(WHO.indexOf(w) >= 0 && t > (last[w] || 0)) last[w] = t; }
  var out = [];
  WHO.forEach(function(w){ var n = schoolDaysSince_(last[w]); if(n >= (min || 3)) out.push({ who: w, days: n >= 99 ? 0 : n }); });
  return out;
}

/* ==========================================================
   Version 5: the Cultural Calendar school count (/cc/).
   Anyone signed in to a school Google account can add their go
   to the count; the page shows totals only. The CC counts tab
   stores a scrambled tag for each person, never a name or email,
   so the same person on two devices still only counts once.
   The tab is made the first time someone has a go: no need to
   run setup again, just deploy a new version.
   ========================================================== */
var CCH = ["Week", "Item", "Person", "On", "Updated"];
function ccTab_(){
  var ss = SpreadsheetApp.getActive(), sh = ss.getSheetByName("CC counts");
  if(sh) return sh;
  sh = tab_(ss, "CC counts", CCH);
  sh.getRange("A:C").setNumberFormat("@");
  sh.getRange(1, CCH.length + 2).setValue("Written by the Cultural Calendar page. Person is a scrambled tag, not a name. Leave it alone.");
  return sh;
}
function ccPerson_(){
  var email = "";
  try{ email = String(Session.getActiveUser().getEmail() || "").trim().toLowerCase(); }catch(e){}
  if(!email) throw new Error("not signed in");
  return ccHash_(email);
}
/* a salted one way tag: the salt lives in the script's own properties,
   so nobody can work back from the tag to an email */
function ccHash_(email){
  var props = PropertiesService.getScriptProperties(), salt = props.getProperty("ccSalt");
  if(!salt){ salt = randomKey_() + randomKey_(); props.setProperty("ccSalt", salt); }
  var d = Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, salt + "|" + email, Utilities.Charset.UTF_8), h = "";
  for(var i = 0; i < 8; i++){ var b = (d[i] + 256) % 256; h += (b < 16 ? "0" : "") + b.toString(16); }
  return h;
}
function ccArgs_(r){
  var week = String(r.week || ""), item = String(r.item || "");
  if(!/^[a-z0-9-]{2,40}$/.test(week)) throw new Error("bad week");
  if(item && !/^(t[0-9]|game)$/.test(item)) throw new Error("bad item");
  return { week: week, item: item };
}
function ccCounts_(r){
  var a = ccArgs_(r), me = "";
  try{ me = ccPerson_(); }catch(e){}
  var cache = CacheService.getScriptCache(), key = "cc.v1." + a.week, hit = cache.get(key), base = null;
  if(hit){ try{ base = JSON.parse(hit); }catch(e0){} }
  var mine = [], mineAll = {};
  if(!base || me){
    var v = rows_("CC counts"), counts = {}, people = {};
    for(var i = 1; i < v.length; i++){
      if(!ticked_(v[i][3])) continue;
      var wk = String(v[i][0]), mineRow = me && String(v[i][2]) === me;
      /* everything this person has on, every week, so a new device can catch up */
      if(mineRow && r.all){ (mineAll[wk] = mineAll[wk] || []).push(String(v[i][1])); }
      if(wk !== a.week) continue;
      counts[v[i][1]] = (counts[v[i][1]] || 0) + 1;
      people[v[i][2]] = 1;
      if(mineRow) mine.push(String(v[i][1]));
    }
    if(!base){ base = { counts: counts, people: Object.keys(people).length }; try{ cache.put(key, JSON.stringify(base), 60); }catch(e1){} }
  }
  var out = { ok: true, week: a.week, counts: base.counts, people: base.people, mine: mine, signedIn: !!me };
  if(r.all && me) out.mineAll = mineAll;
  return out;
}
function ccTick_(r){
  var a = ccArgs_(r);
  if(!a.item) throw new Error("bad item");
  var me = ccPerson_(), on = r.on === true, sh = ccTab_(), v = sh.getDataRange().getValues(), now = new Date(), n = 0;
  for(var j = 1; j < v.length; j++) if(String(v[j][2]) === me) n++;
  for(var i = 1; i < v.length; i++){
    if(String(v[i][0]) === a.week && String(v[i][1]) === a.item && String(v[i][2]) === me){
      sh.getRange(i + 1, 4, 1, 2).setValues([[on, now]]);
      try{ CacheService.getScriptCache().remove("cc.v1." + a.week); }catch(e){}
      return ccCounts_(r);
    }
  }
  /* a year is about 35 weeks of up to 6 things each; anything past that is not a real person having goes */
  if(on && n >= 260) throw new Error("too many");
  if(on) sh.appendRow([a.week, a.item, me, true, now]);
  try{ CacheService.getScriptCache().remove("cc.v1." + a.week); }catch(e2){}
  return ccCounts_(r);
}
READ3.cccounts = ccCounts_;
WRITE3.cctick = ccTick_;
