/* ==========================================================
   Base 5 - the shared memory
   Paste all of this into Extensions > Apps Script in a Google
   Sheet in the school Drive. Run setup once, then deploy it as a
   web app (Execute as: Me, Who has access: Anyone). The site at
   cbsbase5.github.io talks to the web app address.

   Tabs it makes:
     Settings  the staff key, and whether sends are emailed to you
     Folders   type a row here and it shows in that student's folder
               (tick Hide to take something out without deleting it)
     Sends     everything students send to Jay, newest at the bottom
     Progress  the site writes this one; leave it alone

   Nothing here asks for names: students are their initials.
   ========================================================== */

var WHO = ["EGW", "LP", "LW", "LC", "HK", "JT"];
var MAX_CELL = 45000;

/* ---------- run this once ---------- */
function setup(){
  var ss = SpreadsheetApp.getActive();
  var me = "";
  try{ me = Session.getEffectiveUser().getEmail(); }catch(e){}

  var st = tab_(ss, "Settings", ["Setting", "Value", "What it does"]);
  if(st.getLastRow() < 2){
    st.getRange(2, 1, 3, 3).setValues([
      ["Staff key", randomKey_(), "Type this into the live view (cbsbase5.github.io/teacher/live.html) to see everything. Change it to anything long if you like."],
      ["Email me sends", "TRUE", "TRUE sends you an email each time a student sends something. FALSE turns that off."],
      ["Email address", me, "Where those emails go."]
    ]);
  }
  st.setColumnWidth(1, 160); st.setColumnWidth(2, 260); st.setColumnWidth(3, 520);

  var fo = tab_(ss, "Folders", ["Who", "Type", "Date", "Title", "Text or link", "Note", "Hide"]);
  if(fo.getLastRow() < 2) seedFolders_(fo);
  var rows = Math.max(fo.getMaxRows() - 1, 200);
  fo.getRange(2, 1, rows, 1).setDataValidation(SpreadsheetApp.newDataValidation().requireValueInList(WHO.concat(["Everyone"]), true).build());
  fo.getRange(2, 2, rows, 1).setDataValidation(SpreadsheetApp.newDataValidation().requireValueInList(["Message", "Something to try", "File"], true).build());
  fo.getRange(2, 3, rows, 1).setNumberFormat("yyyy-mm-dd");
  fo.getRange(2, 7, rows, 1).setDataValidation(SpreadsheetApp.newDataValidation().requireCheckbox().build());
  fo.setColumnWidth(4, 220); fo.setColumnWidth(5, 420); fo.setColumnWidth(6, 300);

  var se = tab_(ss, "Sends", ["Time", "Who", "Page", "What", "Text", "Seen"]);
  se.getRange("B:E").setNumberFormat("@");
  se.setColumnWidth(5, 520);

  var pr = tab_(ss, "Progress", ["Who", "Key", "Value", "Updated"]);
  pr.getRange("A:C").setNumberFormat("@");
  pr.getRange(1, 6).setValue("The site writes this tab. Leave it alone.");

  ss.setActiveSheet(fo);
  return "Done. Now deploy it as a web app.";
}

function tab_(ss, name, head){
  var sh = ss.getSheetByName(name) || ss.insertSheet(name);
  if(String(sh.getRange(1, 1).getValue()) !== head[0]){
    sh.getRange(1, 1, 1, head.length).setValues([head]).setFontWeight("bold");
    sh.setFrozenRows(1);
  }
  return sh;
}

function randomKey_(){
  var c = "abcdefghjkmnpqrstuvwxyz23456789", s = "";
  for(var i = 0; i < 16; i++) s += c.charAt(Math.floor(Math.random() * c.length));
  return s;
}

/* what is in the folders already, so nothing goes missing on the switch */
function seedFolders_(fo){
  var today = new Date(2026, 9, 18), rows = [];
  WHO.forEach(function(w){
    rows.push([w, "Message", today, "", "Hello " + w + ". This is your folder. Anything I put in here is just for you: things to try, files, and the odd message like this one.\n\nThere is no rush, and nothing in here is marked. If you ever want to say something back, there is a box just under this.", "", false]);
    if(w === "LW") rows.push([w, "Something to try", "", "Box-man: Unboxed!", "/lw/boxman/", "Your game, made from your film. You are the creative director, so play it, break it, and if anything should be different, say the word. There is an intro comic, Box-man mode once you finish a run, and a room designer in Extras.", false]);
    rows.push([w, "Something to try", "", "Idea Lab", "/projects/ideas.html", "For the new project, if you fancy it. Spin a time machine, sort some should we? dilemmas and keep anything that sounds good. Nothing you pick is final.", false]);
  });
  fo.getRange(2, 1, rows.length, 7).setValues(rows);
}

/* ---------- the web app ---------- */
function doGet(e){
  var p = (e && e.parameter) || {};
  try{
    if(p.action === "ping") return out_({ ok: true });
    if(p.action === "load") return out_(load_(who_(p.who), +p.since || 0));
    if(p.action === "folder") return out_(folder_(who_(p.who)));
    if(p.action === "overview") return out_(overview_(p.key));
    return out_({ ok: false, error: "unknown action" });
  }catch(err){ return out_({ ok: false, error: String(err.message || err) }); }
}

function doPost(e){
  var b = {};
  try{ b = JSON.parse((e && e.postData && e.postData.contents) || "{}"); }catch(err){ return out_({ ok: false, error: "bad body" }); }
  var lock = LockService.getScriptLock();
  try{
    lock.waitLock(15000);
    if(b.action === "save") return out_(save_(who_(b.who), b.items || {}));
    if(b.action === "send") return out_(send_(b));
    if(b.action === "seen") return out_(seen_(b.key, +b.row, b.seen !== false));
    return out_({ ok: false, error: "unknown action" });
  }catch(err){
    return out_({ ok: false, error: String(err.message || err) });
  }finally{
    try{ lock.releaseLock(); }catch(e2){}
  }
}

function out_(o){
  return ContentService.createTextOutput(JSON.stringify(o)).setMimeType(ContentService.MimeType.JSON);
}
function who_(w){
  w = String(w || "").toUpperCase();
  if(WHO.indexOf(w) < 0) throw new Error("unknown student");
  return w;
}
function sheet_(name){ return SpreadsheetApp.getActive().getSheetByName(name); }
function setting_(name){
  var v = sheet_("Settings").getDataRange().getValues();
  for(var i = 1; i < v.length; i++) if(String(v[i][0]).trim() === name) return v[i][1];
  return "";
}
/* text that starts like a formula is kept as plain text */
function plain_(s){
  s = String(s == null ? "" : s);
  return /^[=+\-@]/.test(s) ? "'" + s : s;
}

/* ---------- progress ---------- */
function load_(who, since){
  var v = sheet_("Progress").getDataRange().getValues(), items = {};
  for(var i = 1; i < v.length; i++){
    if(v[i][0] !== who) continue;
    var t = +v[i][3] || 0;
    if(t > since) items[v[i][1]] = { v: String(v[i][2]), t: t };
  }
  return { ok: true, items: items, now: Date.now() };
}

function save_(who, items){
  var sh = sheet_("Progress"), v = sh.getDataRange().getValues(), at = {};
  for(var i = 1; i < v.length; i++) if(v[i][0] === who) at[v[i][1]] = i + 1;
  var add = [], changed = 0;
  Object.keys(items).forEach(function(k){
    var it = items[k] || {}, val = it.v == null ? "" : String(it.v), t = +it.t || Date.now();
    if(k.length > 200 || val.length > MAX_CELL) return;
    var r = at[k];
    if(r){
      if(t < (+v[r - 1][3] || 0)) return;            /* something newer is already here */
      sh.getRange(r, 3, 1, 2).setValues([[val, t]]);
    } else add.push([who, k, val, t]);
    changed++;
  });
  if(add.length) sh.getRange(sh.getLastRow() + 1, 1, add.length, 4).setValues(add);
  return { ok: true, saved: changed, now: Date.now() };
}

/* ---------- sends to Jay ---------- */
function send_(b){
  var who = String(b.who || "").toUpperCase().replace(/[^A-Z0-9 ]/g, "").slice(0, 8) || "?";
  var what = String(b.tool || "Something").slice(0, 120);
  var page = String(b.page || "").slice(0, 200);
  var text = String(b.text || "").slice(0, MAX_CELL);
  var sh = sheet_("Sends");
  sh.appendRow([new Date(), plain_(who), plain_(page), plain_(what), plain_(text), ""]);
  try{
    var on = String(setting_("Email me sends")).toUpperCase() === "TRUE", to = String(setting_("Email address")).trim();
    if(on && to) MailApp.sendEmail(to, "Base 5: " + who + " sent " + what, text + "\n\n(From " + page + ". Everything is in the Sends tab of your Base 5 sheet.)");
  }catch(err){}
  return { ok: true };
}

function seen_(key, row, seen){
  staff_(key);
  var sh = sheet_("Sends");
  if(row < 2 || row > sh.getLastRow()) throw new Error("no such row");
  sh.getRange(row, 6).setValue(seen ? "yes" : "");
  return { ok: true };
}

/* ---------- folders ---------- */
function folder_(who){
  var v = sheet_("Folders").getDataRange().getValues(), tz = Session.getScriptTimeZone();
  var out = { messages: [], activities: [], files: [] };
  for(var i = 1; i < v.length; i++){
    var r = v[i], w = String(r[0]).trim();
    if(w !== who && w !== "Everyone") continue;
    if(r[6] === true || String(r[6]).toUpperCase() === "TRUE") continue;   /* ticked Hide */
    var type = String(r[1]).trim(), title = String(r[3]).trim(), text = String(r[4]).trim(), note = String(r[5]).trim();
    if(!type || (!title && !text)) continue;
    var date = r[2] instanceof Date ? Utilities.formatDate(r[2], tz, "yyyy-MM-dd") : String(r[2] || "").trim();
    var id = "r" + digest_(w + "|" + type + "|" + title + "|" + text);
    if(type === "Message") out.messages.push({ id: id, date: date, text: text || title });
    else if(type === "File") out.files.push({ id: id, title: title || "A file", href: text, note: note });
    else out.activities.push({ id: id, title: title || "Something to try", href: text, note: note });
  }
  return { ok: true, who: who, data: out, now: Date.now() };
}

function digest_(s){
  var d = Utilities.computeDigest(Utilities.DigestAlgorithm.MD5, s, Utilities.Charset.UTF_8), h = "";
  for(var i = 0; i < 6; i++){ var b = (d[i] + 256) % 256; h += (b < 16 ? "0" : "") + b.toString(16); }
  return h;
}

/* ---------- Jay's live view ---------- */
function staff_(key){
  var real = String(setting_("Staff key")).trim();
  if(!real || String(key || "").trim() !== real) throw new Error("staff key");
}

function overview_(key){
  staff_(key);
  var v = sheet_("Progress").getDataRange().getValues(), students = {};
  WHO.forEach(function(w){ students[w] = { items: {}, last: 0 }; });
  for(var i = 1; i < v.length; i++){
    var s = students[v[i][0]]; if(!s) continue;
    var t = +v[i][3] || 0;
    s.items[v[i][1]] = { v: String(v[i][2]), t: t };
    if(t > s.last) s.last = t;
  }
  var sv = sheet_("Sends").getDataRange().getValues(), sends = [];
  for(var j = Math.max(1, sv.length - 300); j < sv.length; j++){
    var r = sv[j];
    sends.push({ row: j + 1, time: r[0] instanceof Date ? r[0].getTime() : String(r[0]), who: String(r[1]), page: String(r[2]), what: String(r[3]), text: String(r[4]), seen: String(r[5]) === "yes" });
  }
  return { ok: true, students: students, sends: sends, now: Date.now() };
}
