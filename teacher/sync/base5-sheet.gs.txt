/* ==========================================================
   Base 5 - the shared memory
   Paste all of this into Extensions > Apps Script in a Google
   Sheet in the school Drive. Run setup once, then deploy it as a
   web app (Execute as: Me; Who has access: anyone in the school).

   The site at cbsbase5.github.io talks to it through a small
   hidden page (the bridge, ?bridge=1) that opens inside the site
   with the student's own school Google sign-in, so the web app can
   stay school-only. It can also be called directly, which only
   works when access is set to Anyone.

     People    each student's and staff member's school email, and
               each student's code; signed in, pages open by themselves
     Settings  the staff passphrase, and whether sends are emailed to you

               (tick Hide to take something out without deleting it;
               a Date still to come keeps it back until that day)
     Sends     everything students send to Jay, newest at the bottom
     Progress  the site writes this one; leave it alone

   Nothing here asks for names: students are their initials.
   ========================================================== */

function doPost(e){
  var b = {};

/* every request, from the bridge or called directly */
function api(r){
  r = r || {};
  try{
    var a = r.action;
    if(a === "ping") return { ok: true };
    if(a === "load") return load_(who_(r.who), +r.since || 0);
    if(a === "folder") return folder_(who_(r.who));
    if(a === "overview") return overview_(r.key);
    if(a === "whoami") return whoami_();
    if(a === "codes") return codes_(r.key);
    var read3 = READ3[a];
    if(read3) return read3(r);
    if(WRITE3[a]){
      var lock3 = LockService.getScriptLock();
      lock3.waitLock(15000);
      try{ return WRITE3[a](r); }finally{ try{ lock3.releaseLock(); }catch(e3){} }
    }
    if(a === "save" || a === "send" || a === "seen"){
      var lock = LockService.getScriptLock();
      lock.waitLock(15000);
      try{
        if(a === "save") return save_(who_(r.who), r.items || {});
        if(a === "send") return send_(r);
        return seen_(r.key, +r.row, r.seen !== false);
      }finally{ try{ lock.releaseLock(); }catch(e2){} }
    }
    return { ok: false, error: "unknown action" };
  }catch(err){ return { ok: false, error: String(err.message || err) }; }
}
/* the bridge page calls this with google.script.run; text in, text out */
function apiJson(req){
  var r = {};
  try{ r = typeof req === "string" ? JSON.parse(req) : (req || {}); }catch(e){}
  return JSON.stringify(api(r));
}

function bridgeHtml_(){
  return '<!DOCTYPE html><html><head><meta charset="utf-8"></head><body><script>' +
    'var SITE = ' + JSON.stringify(SITE) + ';' +
    'window.addEventListener("message", function(e){' +
    '  if(e.origin !== SITE) return;' +
    '  var m = e.data; if(!m || m.b5 !== "req") return;' +
    '  var reply = function(body){ e.source.postMessage({ b5: "res", id: m.id, body: body }, e.origin); };' +
    '  google.script.run' +
    '    .withSuccessHandler(reply)' +
    '    .withFailureHandler(function(err){ reply(JSON.stringify({ ok: false, error: String((err && err.message) || err) })); })' +
    '    .apiJson(JSON.stringify(m.req || {}));' +
    '});' +
    'try{ window.top.postMessage({ b5: "ready" }, SITE); }catch(x){}' +
    '<\/script></body></html>';
}

/* ---------- progress ---------- */
function load_(who, since){
  var v = sheet_("Progress").getDataRange().getValues(), items = {};
  for(var i = 1; i < v.length; i++){
    if(v[i][0] !== who) continue;
    var t = +v[i][3] || 0;
    if(t > since) items[v[i][1]] = { v: String(v[i][2]), t: t };
  }
  return { ok: true, items: items, now: Date.now() };
}

function save_(who, items){
  var sh = sheet_("Progress"), v = sh.getDataRange().getValues(), at = {};
  for(var i = 1; i < v.length; i++) if(v[i][0] === who) at[v[i][1]] = i + 1;
  try{ fuelBaseline_(who, v); }catch(e0){}
  var events = [];
  var add = [], changed = 0;
  Object.keys(items).forEach(function(k){
    var it = items[k] || {}, val = it.v == null ? "" : String(it.v), t = +it.t || Date.now();
    if(k.length > 200 || val.length > MAX_CELL) return;
    var r = at[k];
    if(r){
      if(t < (+v[r - 1][3] || 0)) return;            /* something newer is already here */
      try{ events = events.concat(events_(who, k, String(v[r - 1][2]), val)); }catch(e1){}
      sh.getRange(r, 3, 1, 2).setValues([[val, t]]);
    } else {
      try{ events = events.concat(events_(who, k, "", val)); }catch(e2){}
      add.push([who, k, val, t]);
    }
    changed++;
  });
  if(add.length) sh.getRange(sh.getLastRow() + 1, 1, add.length, 4).setValues(add);
  if(events.length) logEvents_(who, events);
  return { ok: true, saved: changed, now: Date.now() };
}

/* ---------- sends to Jay ---------- */
function send_(b){
  var who = String(b.who || "").toUpperCase().replace(/[^A-Z0-9 ]/g, "").slice(0, 8) || "?";
  var what = String(b.tool || "Something").slice(0, 120);
  var page = String(b.page || "").slice(0, 200);
  var text = String(b.text || "").slice(0, MAX_CELL);
  var sh = sheet_("Sends");
  sh.appendRow([new Date(), plain_(who), plain_(page), plain_(what), plain_(text), ""]);
  if(WHO.indexOf(who) >= 0) logEvents_(who, [["Sent to Jay", what]]);
  if(what === "Idea for the site"){
    var idea = text.replace(/^An idea for the Base 5 site:\s*/, "").trim();
    if(idea && idea !== "(nothing typed yet)") sheet_("Ideas").appendRow([new Date(), plain_(idea.slice(0, 500)), "Thinking about it", "", 0, "", false]);
  }
  try{
    var on = String(setting_("Email me sends")).toUpperCase() === "TRUE", to = String(setting_("Email address")).trim();
    if(on && to) MailApp.sendEmail(to, "Base 5: " + who + " sent " + what, text + "\n\n(From " + page + ". Everything is in the Sends tab of your Base 5 sheet.)");
  }catch(err){}
  return { ok: true };
}

/* ---------- folders ---------- */
function folder_(who){
  var v = sheet_("Folders").getDataRange().getValues(), tz = Session.getScriptTimeZone();
  var out = { messages: [], activities: [], files: [], a2: false };
  var today = Utilities.formatDate(new Date(), tz, "yyyy-MM-dd");
  for(var i = 1; i < v.length; i++){
    var r = v[i], w = String(r[0]).trim();
    if(w !== who && w !== "Everyone") continue;
    var type = String(r[1]).trim(), title = String(r[3]).trim(), text = String(r[4]).trim(), note = String(r[5]).trim();
    if(/\/a2-[a-z0-9-]+\.html/.test(text)) out.a2 = true;      /* the Autumn 2 plan is in the sheet */
    if(r[6] === true || String(r[6]).toUpperCase() === "TRUE") continue;   /* ticked Hide */
    if(!type || (!title && !text)) continue;
    var date = r[2] instanceof Date ? Utilities.formatDate(r[2], tz, "yyyy-MM-dd") : String(r[2] || "").trim();
    if(date && date > today) continue;                            /* a date still to come: it waits until then */
    var id = "r" + digest_(w + "|" + type + "|" + title + "|" + text);
    if(type === "Message") out.messages.push({ id: id, date: date, text: text || title });
    else if(type === "File") out.files.push({ id: id, title: title || "A file", href: text, note: note, date: date });
    else out.activities.push({ id: id, title: title || "Something to try", href: text, note: note, date: date });
  }
  return { ok: true, who: who, data: out, now: Date.now() };
}

/* ---------- Autumn 2: put the term's drops into the Folders tab ----------
   Staff only. rows: [{who, date, title, href, note}]. Adds each row that
   is not there already (same student and same page), dated, so it shows
   in the folder from that day. Safe to run again. */
function a2plan_(r){
  staff_(r.key);
  var rows = Array.isArray(r.rows) ? r.rows : [];
  var sh = sheet_("Folders"), v = sh.getDataRange().getValues(), have = {};
  function base(t){ return String(t || "").trim().split("?")[0]; }
  for(var i = 1; i < v.length; i++) have[String(v[i][0]).trim() + "|" + base(v[i][4])] = 1;
  var add = [];
  rows.forEach(function(x){
    var who = String(x.who || "").toUpperCase(), href = String(x.href || "").trim();
    if(WHO.indexOf(who) < 0 || !/^\/[a-z0-9\-\/]+\/a2-[a-z0-9-]+\.html(\?lv=[a-z]+)?$/.test(href)) return;
    if(!/^\d{4}-\d{2}-\d{2}$/.test(String(x.date || ""))) return;
    if(have[who + "|" + base(href)]) return;
    have[who + "|" + base(href)] = 1;
    var p = String(x.date).split("-");
    add.push([who, "Something to try", new Date(+p[0], +p[1] - 1, +p[2]), clean_(x.title, 120), href, clean_(x.note || "", 600), false]);
  });
  if(add.length) sh.getRange(sh.getLastRow() + 1, 1, add.length, 7).setValues(add);
  return { ok: true, added: add.length, skipped: rows.length - add.length };
}

function digest_(s){
  var d = Utilities.computeDigest(Utilities.DigestAlgorithm.MD5, s, Utilities.Charset.UTF_8), h = "";
  for(var i = 0; i < 6; i++){ var b = (d[i] + 256) % 256; h += (b < 16 ? "0" : "") + b.toString(16); }
  return h;
}

/* ---------- events (the timeline) ---------- */
function events_(who, key, oldV, newV){
  var P = PREFIX[who], out = [];
  if(!P || key.indexOf(P) !== 0) return out;
  var id = key.slice(P.length), o = parse_(oldV), n = parse_(newV);
